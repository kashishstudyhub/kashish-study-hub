1. Data Types in C

A data type tells the compiler what type of value a variable can store. It also determines the operations that can be performed on that data.

Data Type Use Example
int Integer values 10, -5, 100
char Single character 'A', 'b'
float Decimal values 10.5
double More precision for decimal values 25.6789

Examples

int age = 20;

char grade = 'A';

float price = 25.50f;

double value = 123.456789;
Remember:

A character is written in single quotes, such as 'A', while a string is written in double quotes, such as "Hello".

2. Variables

A variable is a named memory location used to store a value that can change during program execution.

Syntax

data_type variable_name;

Example

int age;

age = 20;

Declaration and Initialization

int marks = 85;

float height = 5.5f;

char grade = 'A';

Rules for Variable Names

  • Must begin with a letter or underscore.
  • Cannot contain spaces.
  • Cannot be a C keyword.
  • C is case-sensitive.
  • Digits can be used after the first character.

3. Constants

A constant is a value that does not change during program execution.

Using const

const float PI = 3.14159f;

The value of a const object should not be modified through that object.

4. printf() and scanf()

printf()

The printf() function is used to display formatted output on the screen. 

#include <stdio.h>

int main()
{
    int age = 20;

    printf("Age = %d", age);

    return 0;
}

Common Format Specifiers

Specifier Used For
%d int
%f floating-point output
%c char
%s string

scanf()

The scanf() function is commonly used to read formatted input. 

#include <stdio.h>

int main()
{
    int age;

    printf("Enter age: ");
    scanf("%d", &age);

    printf("Age = %d", age);

    return 0;
}
Important:

For an ordinary scalar variable such as int, scanf() generally receives its address using the & operator.

5. Arithmetic Operators

Arithmetic operators are used to perform mathematical operations.

Operator Meaning Example
+ Addition a + b
- Subtraction a - b
* Multiplication a * b
/ Division a / b
% Remainder a % b

Example

#include <stdio.h>

int main()
{
    int a = 10;
    int b = 3;

    printf("Addition = %d\n", a + b);
    printf("Subtraction = %d\n", a - b);
    printf("Multiplication = %d\n", a * b);
    printf("Division = %d\n", a / b);
    printf("Remainder = %d\n", a % b);

    return 0;
}

6. Relational Operators

Relational operators compare two values. The result of a comparison is used as a true/false condition.

Operator Meaning
== Equal to
!= Not equal to
> Greater than
< Less than
>= Greater than or equal to
<= Less than or equal to

Example

int a = 10;
int b = 20;

printf("%d", a < b);

7. Logical Operators

Logical operators combine or invert conditions.

Operator Meaning
&& Logical AND
|| Logical OR
! Logical NOT

Example

#include <stdio.h>

int main()
{
    int age = 20;

    if(age >= 18 && age <= 60)
    {
        printf("Condition is true");
    }

    return 0;
}

8. Increment and Decrement Operators

The increment operator ++ increases a value by 1. The decrement operator -- decreases a value by 1.

Increment

int a = 5;

a++;

printf("%d", a);

Decrement

int a = 5;

a--;

printf("%d", a);

Prefix and Postfix

++a;   /* prefix */

a++;   /* postfix */

--a;   /* prefix */

a--;   /* postfix */

9. if-else Condition

The if statement executes a block when its condition is true. The else block executes when the condition is false.

Syntax

if(condition)
{
    // statements
}
else
{
    // statements
}

Example: Even or Odd

#include <stdio.h>

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    if(n % 2 == 0)
    {
        printf("Even number");
    }
    else
    {
        printf("Odd number");
    }

    return 0;
}

10. Nested if

An if statement placed inside another if statement is called a nested if.

Example

#include <stdio.h>

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    if(n > 0)
    {
        if(n % 2 == 0)
        {
            printf("Positive even number");
        }
        else
        {
            printf("Positive odd number");
        }
    }

    return 0;
}

11. for Loop

The for loop is useful when the number of iterations is known or can be controlled with an initialization, condition and update.

Syntax

for(initialization; condition; update)
{
    // statements
}

Example: Print 1 to 10

#include <stdio.h>

int main()
{
    int i;

    for(i = 1; i <= 10; i++)
    {
        printf("%d ", i);
    }

    return 0;
}

12. while Loop

The while loop repeatedly executes its body as long as its condition remains true.

Syntax

while(condition)
{
    // statements
}

Example

#include <stdio.h>

int main()
{
    int i = 1;

    while(i <= 5)
    {
        printf("%d ", i);
        i++;
    }

    return 0;
}

13. do-while Loop

The do-while loop executes its body first and checks the condition afterwards. Therefore, its body executes at least once.

Syntax

do
{
    // statements
}
while(condition);

Example

#include <stdio.h>

int main()
{
    int i = 1;

    do
    {
        printf("%d ", i);
        i++;
    }
    while(i <= 5);

    return 0;
}

14. Important C Programs

Program 1: Find Largest of Two Numbers

#include <stdio.h>

int main()
{
    int a, b;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    if(a > b)
    {
        printf("Largest = %d", a);
    }
    else
    {
        printf("Largest = %d", b);
    }

    return 0;
}

Program 2: Check Positive, Negative or Zero

#include <stdio.h>

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    if(n > 0)
    {
        printf("Positive");
    }
    else if(n < 0)
    {
        printf("Negative");
    }
    else
    {
        printf("Zero");
    }

    return 0;
}

Program 3: Sum of First N Natural Numbers

#include <stdio.h>

int main()
{
    int n, i;
    int sum = 0;

    printf("Enter n: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        sum = sum + i;
    }

    printf("Sum = %d", sum);

    return 0;
}

Program 4: Multiplication Table

#include <stdio.h>

int main()
{
    int n, i;

    printf("Enter number: ");
    scanf("%d", &n);

    for(i = 1; i <= 10; i++)
    {
        printf("%d x %d = %d\n", n, i, n * i);
    }

    return 0;
}

Program 5: Factorial Using Loop

#include <stdio.h>

int main()
{
    int n, i;
    int factorial = 1;

    printf("Enter number: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        factorial = factorial * i;
    }

    printf("Factorial = %d", factorial);

    return 0;
}

Program 6: Check Prime Number

#include <stdio.h>

int main()
{
    int n, i;
    int prime = 1;

    printf("Enter number: ");
    scanf("%d", &n);

    if(n < 2)
    {
        prime = 0;
    }
    else
    {
        for(i = 2; i < n; i++)
        {
            if(n % i == 0)
            {
                prime = 0;
                break;
            }
        }
    }

    if(prime)
    {
        printf("Prime number");
    }
    else
    {
        printf("Not a prime number");
    }

    return 0;
}

16. Quick Revision

  • int: Used for integer values.
  • char: Used for a character.
  • float: Used for floating-point values.
  • Variable: Named object used to store a value that can change.
  • printf(): Used for formatted output.
  • scanf(): Used for formatted input.
  • Arithmetic: +, -, *, /, %
  • Relational: ==, !=, >, <, >=, <=
  • Logical: &&, ||, !
  • Increment: ++
  • Decrement: --
  • if-else: Used for conditional execution.
  • for: Loop with initialization, condition and update.
  • while: Condition is checked before each iteration.
  • do-while: Body executes before the condition is checked.
