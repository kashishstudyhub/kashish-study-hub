1. Introduction to Arrays

An array is a collection of elements of the same data type stored under one name. Each element can be accessed using an index.

Advantages of Arrays

  • Stores multiple values under one name.
  • Makes handling similar data easier.
  • Elements can be accessed using indexes.
  • Useful for storing marks, numbers, tables and other data.
Important:

Array indexing in C starts from 0.

Example

int marks[5];

The indexes of this array are 0, 1, 2, 3 and 4.

2. One-Dimensional Array

A one-dimensional array stores elements in a single sequence.

Syntax

data_type array_name[size];

Example

int marks[5];

Initialization

int marks[5] = {70, 80, 65, 90, 75};

Accessing an Element

printf("%d", marks[0]);

Traversing an Array

int i;

for(i = 0; i < 5; i++)
{
    printf("%d ", marks[i]);
}

3. Important 1D Array Programs

Program 1: Input and Display Array

#include <stdio.h>

int main()
{
    int a[5];
    int i;

    printf("Enter 5 elements:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Array elements are:\n");

    for(i = 0; i < 5; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

Program 2: Sum of Array Elements

#include <stdio.h>

int main()
{
    int a[5];
    int i;
    int sum = 0;

    printf("Enter 5 elements:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
        sum = sum + a[i];
    }

    printf("Sum = %d", sum);

    return 0;
}

Program 3: Largest Element

#include <stdio.h>

int main()
{
    int a[5];
    int i;
    int largest;

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    largest = a[0];

    for(i = 1; i < 5; i++)
    {
        if(a[i] > largest)
        {
            largest = a[i];
        }
    }

    printf("Largest = %d", largest);

    return 0;
}

4. Two-Dimensional Array

A two-dimensional array stores data in rows and columns. It is commonly used for matrices and tables.

Syntax

data_type array_name[rows][columns];

Example

int matrix[2][3];

Initialization

int matrix[2][2] = {
    {1, 2},
    {3, 4}
};

Accessing an Element

printf("%d", matrix[0][1]);

5. Matrix Programs

Program 1: Display a 2 × 2 Matrix

#include <stdio.h>

int main()
{
    int a[2][2];
    int i, j;

    printf("Enter matrix elements:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", a[i][j]);
        }

        printf("\n");
    }

    return 0;
}

Program 2: Matrix Addition

#include <stdio.h>

int main()
{
    int a[2][2], b[2][2], c[2][2];
    int i, j;

    printf("Enter first matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter second matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            c[i][j] = a[i][j] + b[i][j];
        }
    }

    printf("Sum of matrices:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", c[i][j]);
        }

        printf("\n");
    }

    return 0;
}

6. Character Arrays

A character array is an array whose elements have the data type char.

Example

char name[20];

Initialization

char name[] = "Kashish";

Character arrays are commonly used for storing strings.

7. Strings in C

A string in C is a sequence of characters terminated by the null character '\0'.

Example

char name[] = "Kashish";
Important:

The null character '\0' indicates the end of a C string. It is not the same as the character '0'.

String Declaration

char name[50];

8. String Input and Output

Using scanf()

char name[30];

scanf("%29s", name);

printf("%s", name);

A basic %s scanf conversion reads characters until whitespace is encountered.

Using fgets()

#include <stdio.h>

int main()
{
    char name[50];

    printf("Enter your name: ");

    fgets(name, sizeof name, stdin);

    printf("Name: %s", name);

    return 0;
}

9. String Handling Functions

Common string functions are declared in the <string.h> header.

Function Use
strlen() Finds the length of a string.
strcpy() Copies one string into another.
strcat() Appends one string to another.
strcmp() Compares two strings.

strlen()

#include <stdio.h>
#include <string.h>

int main()
{
    char name[] = "Kashish";

    printf("Length = %zu", strlen(name));

    return 0;
}

strcpy()

#include <stdio.h>
#include <string.h>

int main()
{
    char source[] = "Hello";
    char destination[20];

    strcpy(destination, source);

    printf("%s", destination);

    return 0;
}

strcat()

#include <stdio.h>
#include <string.h>

int main()
{
    char first[30] = "Hello ";
    char second[] = "World";

    strcat(first, second);

    printf("%s", first);

    return 0;
}

strcmp()

#include <stdio.h>
#include <string.h>

int main()
{
    char a[] = "apple";
    char b[] = "apple";

    if(strcmp(a, b) == 0)
    {
        printf("Strings are equal");
    }
    else
    {
        printf("Strings are different");
    }

    return 0;
}

10. Important String Programs

Program 1: Find String Length

#include <stdio.h>
#include <string.h>

int main()
{
    char str[100];

    printf("Enter a string: ");

    fgets(str, sizeof str, stdin);

    printf("Length = %zu", strlen(str));

    return 0;
}

Program 2: Copy a String

#include <stdio.h>
#include <string.h>

int main()
{
    char source[100];
    char destination[100];

    printf("Enter string: ");

    fgets(source, sizeof source, stdin);

    strcpy(destination, source);

    printf("Copied string: %s", destination);

    return 0;
}

Program 3: Compare Two Strings

#include <stdio.h>
#include <string.h>

int main()
{
    char first[50];
    char second[50];

    printf("Enter first string: ");
    scanf("%49s", first);

    printf("Enter second string: ");
    scanf("%49s", second);

    if(strcmp(first, second) == 0)
    {
        printf("Strings are equal");
    }
    else
    {
        printf("Strings are different");
    }

    return 0;
}

12. Quick Revision

  • Array: Collection of elements of the same data type.
  • Index: Position used to access an array element.
  • First index: 0.
  • 1D Array: Data arranged in one sequence.
  • 2D Array: Data arranged in rows and columns.

  • String: Character sequence terminated by '\0'.
  • strlen(): Finds string length.
  • strcpy(): Copies a string.
  • strcat(): Appends one string to another.
  • strcmp(): Compares two strings.
