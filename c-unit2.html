1. Operators in C

An operator is a symbol that tells the compiler to perform an operation on one or more operands.

For example, in a + b, the symbols a and b are operands and + is the operator.

Operators Covered in This Unit

  • Arithmetic operators
  • Relational operators
  • Logical operators
  • Increment operator
  • Decrement operator

2. Arithmetic Operators

Arithmetic operators are used to perform common mathematical operations.

Operator Operation Example
+ Addition a + b
- Subtraction a - b
* Multiplication a * b
/ Division a / b
% Remainder a % b

Example

#include <stdio.h>

int main()
{
    int a = 20;
    int b = 6;

    printf("Addition = %d\n", a + b);
    printf("Subtraction = %d\n", a - b);
    printf("Multiplication = %d\n", a * b);
    printf("Division = %d\n", a / b);
    printf("Remainder = %d\n", a % b);

    return 0;
}
Important:

When both operands of / are integers, the result is integer division.

3. Relational Operators

Relational operators compare two values. A comparison expression evaluates to a truth value used by control statements.

Operator Meaning
== Equal to
!= Not equal to
> Greater than
< Less than
>= Greater than or equal to
<= Less than or equal to

Example

#include <stdio.h>

int main()
{
    int a = 10;
    int b = 20;

    printf("%d\n", a < b);
    printf("%d\n", a == b);

    return 0;
}

4. Logical Operators

Logical operators are used to combine or negate conditions.

Operator Name Meaning
&& Logical AND True when both conditions are true.
|| Logical OR True when at least one condition is true.
! Logical NOT Reverses the logical value of its operand.

Example

#include <stdio.h>

int main()
{
    int age = 20;
    int marks = 75;

    if(age >= 18 && marks >= 40)
    {
        printf("Both conditions are satisfied");
    }

    return 0;
}

5. Increment and Decrement Operators

The increment operator ++ increases its operand by one. The decrement operator -- decreases its operand by one.

Increment

int x = 5;

x++;

After the statement, x becomes 6.

Decrement

int x = 5;

x--;

After the statement, x becomes 4.

Prefix and Postfix

Form Example
Prefix ++x
Postfix x++
Prefix decrement --x
Postfix decrement x--
Exam Tip:

In a prefix expression, the increment/decrement occurs before the value is used in that expression. In a postfix expression, the original value is used first and the update occurs afterward.

6. if Statement

The if statement executes a block of statements when its condition is true.

Syntax

if(condition)
{
    statements;
}

Example

#include <stdio.h>

int main()
{
    int number;

    printf("Enter number: ");
    scanf("%d", &number);

    if(number > 0)
    {
        printf("Positive number");
    }

    return 0;
}

7. if-else Statement

The if-else statement provides two alternative paths.

Syntax

if(condition)
{
    statements;
}
else
{
    statements;
}

Example: Even or Odd

#include <stdio.h>

int main()
{
    int n;

    printf("Enter number: ");
    scanf("%d", &n);

    if(n % 2 == 0)
    {
        printf("Even number");
    }
    else
    {
        printf("Odd number");
    }

    return 0;
}

8. else-if Ladder

An else-if ladder is used when multiple conditions need to be checked in sequence.

Example: Grade

#include <stdio.h>

int main()
{
    int marks;

    printf("Enter marks: ");
    scanf("%d", &marks);

    if(marks >= 90)
    {
        printf("Grade A");
    }
    else if(marks >= 75)
    {
        printf("Grade B");
    }
    else if(marks >= 60)
    {
        printf("Grade C");
    }
    else if(marks >= 40)
    {
        printf("Grade D");
    }
    else
    {
        printf("Fail");
    }

    return 0;
}

9. Nested if

A nested if is an if statement placed inside another if statement. It can be useful when a second condition should only be considered after the first condition is satisfied.

Example

#include <stdio.h>

int main()
{
    int age;
    int marks;

    printf("Enter age: ");
    scanf("%d", &age);

    printf("Enter marks: ");
    scanf("%d", &marks);

    if(age >= 18)
    {
        if(marks >= 40)
        {
            printf("Both conditions are satisfied");
        }
    }

    return 0;
}

10. for Loop

The for loop is commonly used when the iteration pattern is known. It contains initialization, condition and update expressions.

Syntax

for(initialization; condition; update)
{
    statements;
}

Example: Print 1 to 10

#include <stdio.h>

int main()
{
    int i;

    for(i = 1; i <= 10; i++)
    {
        printf("%d ", i);
    }

    return 0;
}

Parts of for Loop

Part Purpose
Initialization Sets the initial value.
Condition Controls whether another iteration occurs.
Update Changes the loop control variable.

11. while Loop

The while loop checks its condition before executing the body. Therefore, the body can execute zero times.

Syntax

while(condition)
{
    statements;
}

Example

#include <stdio.h>

int main()
{
    int i = 1;

    while(i <= 5)
    {
        printf("%d ", i);
        i++;
    }

    return 0;
}

12. do-while Loop

The do-while loop executes the body first and checks the condition afterward. Therefore, the body executes at least once.

Syntax

do
{
    statements;
}
while(condition);

Example

#include <stdio.h>

int main()
{
    int i = 1;

    do
    {
        printf("%d ", i);
        i++;
    }
    while(i <= 5);

    return 0;
}

13. break and continue

break

The break statement terminates the nearest enclosing loop or switch. 

for(int i = 1; i <= 10; i++)
{
    if(i == 5)
        break;

    printf("%d ", i);
}

Output: 1 2 3 4

continue

The continue statement skips the remaining part of the current iteration and moves to the next iteration of the loop. 

for(int i = 1; i <= 5; i++)
{
    if(i == 3)
        continue;

    printf("%d ", i);
}

Output: 1 2 4 5

14. Important Programs

Program 1: Find Greater of Two Numbers

#include <stdio.h>

int main()
{
    int a, b;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    if(a > b)
        printf("A is greater");
    else if(b > a)
        printf("B is greater");
    else
        printf("Both are equal");

    return 0;
}

Program 2: Multiplication Table

#include <stdio.h>

int main()
{
    int n, i;

    printf("Enter number: ");
    scanf("%d", &n);

    for(i = 1; i <= 10; i++)
    {
        printf("%d x %d = %d\n", n, i, n * i);
    }

    return 0;
}

Program 3: Sum of Natural Numbers

#include <stdio.h>

int main()
{
    int n, i;
    int sum = 0;

    printf("Enter n: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        sum = sum + i;
    }

    printf("Sum = %d", sum);

    return 0;
}

Program 4: Factorial

#include <stdio.h>

int main()
{
    int n, i;
    long long factorial = 1;

    printf("Enter number: ");
    scanf("%d", &n);

    if(n < 0)
    {
        printf("Factorial is not defined for negative integers.");
        return 0;
    }

    for(i = 1; i <= n; i++)
    {
        factorial = factorial * i;
    }

    printf("Factorial = %lld", factorial);

    return 0;
}

Program 5: Check Positive, Negative or Zero

#include <stdio.h>

int main()
{
    int n;

    printf("Enter number: ");
    scanf("%d", &n);

    if(n > 0)
        printf("Positive");
    else if(n < 0)
        printf("Negative");
    else
        printf("Zero");

    return 0;
}

Program 6: Print Even Numbers

#include <stdio.h>

int main()
{
    int i;

    for(i = 2; i <= 20; i += 2)
    {
        printf("%d ", i);
    }

    return 0;
}
